cmake_minimum_required(VERSION 3.20)
project(FINALPRJ VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# All dependency sources are included: configuration needs no network access.
set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
set(GLFW_LIBRARY_TYPE STATIC CACHE STRING "" FORCE)
add_subdirectory(third_party/glfw EXCLUDE_FROM_ALL)

set(GLM_BUILD_LIBRARY OFF CACHE BOOL "" FORCE)
set(GLM_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(GLM_BUILD_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/glm EXCLUDE_FROM_ALL)

add_library(glad STATIC third_party/glad/src/gl.c)
target_include_directories(glad SYSTEM PUBLIC third_party/glad/include)

add_library(stb_image STATIC src/stb_image.cpp)
target_include_directories(stb_image SYSTEM PUBLIC third_party/stb)

add_library(combat STATIC src/combat.cpp)
target_include_directories(combat PUBLIC src)
target_link_libraries(combat PUBLIC glm::glm)
add_executable(finalprj src/main.cpp src/architecture.cpp src/props.cpp)
target_compile_definitions(finalprj PRIVATE GLFW_INCLUDE_NONE)
target_link_libraries(finalprj PRIVATE glfw glad glm::glm stb_image combat)
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT finalprj)
add_custom_command(TARGET finalprj POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
        ${CMAKE_CURRENT_SOURCE_DIR}/assets $<TARGET_FILE_DIR:finalprj>/assets)

if(MSVC)
    target_compile_options(finalprj PRIVATE /W4 /utf-8)
else()
    target_compile_options(finalprj PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(combat_test tests/combat_test.cpp)
    target_link_libraries(combat_test PRIVATE combat)
    add_test(NAME tactical_combat COMMAND combat_test)
    add_executable(prop_footprint_test tests/prop_footprint_test.cpp)
    target_include_directories(prop_footprint_test PRIVATE src)
    target_link_libraries(prop_footprint_test PRIVATE glm::glm)
    add_test(NAME prop_footprints COMMAND prop_footprint_test)
    add_executable(animation_test tests/animation_test.cpp src/props.cpp)
    target_include_directories(animation_test PRIVATE src)
    target_link_libraries(animation_test PRIVATE glm::glm)
    add_test(NAME hero_animation COMMAND animation_test)
    add_executable(collision_test tests/collision_test.cpp)
    target_include_directories(collision_test PRIVATE src)
    target_link_libraries(collision_test PRIVATE glm::glm)
    add_test(NAME room_collision COMMAND collision_test)
    add_test(NAME crypt_room COMMAND finalprj --smoke-test)
    set_tests_properties(crypt_room PROPERTIES TIMEOUT 60)
endif()
